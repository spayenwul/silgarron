       238 +      def _generate_ridge_mask(self) -> np.ndarray:
       239 +          """
       240 +          Генерирует маску хребта (вертикальная ось "позвоночника").
       241 +  
       242 +          Использует гауссову функцию для создания центральной полосы
       243 +          повышенных значений.
       244 +  
       245 +          Returns:
       246 +              np.ndarray (height, width) - интенсивность хребта [0, 1]
       247 +          """
       248 +          ridge = np.zeros((self.height, self.width), dtype=np.float32)
       249 +  
       250 +          center_x = self.width // 2  # 128 для карты 256×256
       251 +  
       252 +          # Создаём сетку X координат
       253 +          x_coords = np.arange(self.width)
       254 +  
       255 +          # Расстояние каждой точки от центральной оси
       256 +          distance = np.abs(x_coords - center_x)
       257 +  
       258 +          # Нормализуем (0 = центр, 1 = край)
       259 +          normalized_dist = distance / (self.width / 2)
       260 +  
       261 +          # Гауссова функция: e^(-k * dist^2)
       262 +          # k=5 даёт узкий хребет (~20% ширины карты)
       263 +          ridge_strength = np.exp(-5 * normalized_dist**2)
       264 +  
       265 +          # Применяем к каждой строке (одинаково для всей высоты)
       266 +          ridge[:, :] = ridge_strength[np.newaxis, :]
       267 +  
       268 +          return ridge

       273 +  
       274 +          Использует косинусную волну с затуханием от центра для создания
       275 +          "рёбер", отходящих от позвоночника.
       276 +

       278 +              np.ndarray (height, width) - интенсивность рёбер [0, ~0.3]
       279 +          """

       281 +  
       282 +          center_x = self.width // 2

       300 +          rib_wave = np.maximum(0, np.cos(rib_phase * 2 * np.pi))

       304 +          decay = np.exp(-2 * distance_from_ridge**2)

> Детальный анализ графика №2 (Heatmaps)
Это самый важный график для оценки.
Панель 1. Base Elevation (Perlin Noise)
Оценка: ✅ Отлично. Это стандартный, хорошо выглядящий шум Перлина. Он создает прекрасную "текстуру" микрорельефа,

Панель 2. Ridge Mask (Хребет)
Оценка: 🟡 Требует доработки.
Ваш вопрос: "Норм это для хребта гор или нет?"
Ответ: Для геологического, естественного горного хребта — категорически нет. Он слишком идеальный: идеально прямой, 
идеально вертикальный, с идеально симметричным градиентом. В природе таких форм не бывает.
Ответ в контексте лора Сильгаррона: Для анатомического "позвоночника" мира — это уже ближе к цели, так как 
позвоночник по своей сути более регулярен, чем горный массив. НО, даже для позвоночника это слишком идеально.
Ключевая проблема: В этом графике отсутствуют "изгибы" (wiggle), которые упоминаются в техническом плане спринта 
(Задача 1.2, функция _create_spine_mask). Там есть строка wiggle = int(np.sin(y / 20.0) * 10). Судя по графику, этот
 изгиб либо не применяется, либо его амплитуда равна нулю. Именно он должен был сломать идеальную вертикальную линию
 и придать хребту более органичный, змеевидный вид.
Панель 3. Rib Mask (Рёбра)
Оценка: 🟡 Требует доработки.
Проблема: Та же, что и у хребта. "Рёбра" — это идеально прямые диагональные линии с одинаковым расстоянием между 
ними. Настоящие рёбра имеют изгиб и не являются абсолютно параллельными. Это вносит в карту второй слой 
искусственности.
Панель 4. Final Elevation (Финальная Высота)
Оценка: 🟡 Результат предсказуем, но не идеален.
Что мы видим: Органическая текстура шума Перлина (из панели 1) была наложена на очень искусственный "каркас" (из 
панелей 2 и 3). В итоге, хребет все еще выглядит как ровная вертикальная полоса, просто "затекстурированная" 
неровностями. Диагональная структура рёбер также слегка просматривается.
Несоответствие с планом: Заголовок гласит "30% Base + 50% Ridge + 20% Ribs". В плане спринта (Задача 1.2) указаны 
другие веса: "60% базовый шум + 30% хребет + 10% рёбра". Вес хребта в 50% слишком велик — он "задавливает" 
естественность базового шума. Пропорции из документации дали бы более сбалансированный и естественный результат.
